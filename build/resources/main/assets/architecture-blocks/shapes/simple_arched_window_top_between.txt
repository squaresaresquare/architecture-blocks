public VoxelShape makeShape(){
	VoxelShape shape = Shapes.empty();
	shape = Shapes.join(shape, Shapes.box(0, 0, 0, 0.9375, 1, 0.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.25, 0, 0.75, 0.6875, 0.1875, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.1875, 0.1875, 0.75, 0.75, 0.375, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.125, 0.375, 0.75, 0.8125, 0.5625, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.0625, 0.5625, 0.75, 0.875, 0.625, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0, 0.625, 0.75, 0.9375, 0.6875, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0, 0.6875, 0.75, 0.9375, 1, 1), BooleanOp.OR);

	return shape;
}
